cmake_minimum_required(VERSION 3.8)
project(lab_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(lab_interfaces REQUIRED)
add_executable(monitor src/monitor.cpp)
target_compile_features(monitor PUBLIC cxx_std_17)
target_compile_options(monitor PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(monitor rclcpp lab_interfaces)
install(TARGETS monitor DESTINATION lib/${PROJECT_NAME})
ament_package()
